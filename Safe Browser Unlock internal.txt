    const cmContainer = active?.closest?.('.CodeMirror, .cm-editor') || win.document.querySelector('.CodeMirror, .cm-editor');
    if (cmContainer) {
      if (cmContainer.CodeMirror) {
        const cm = cmContainer.CodeMirror;
        try {
          if (typeof cm.setOption === 'function' && cm.getOption('readOnly')) {
            cm.setOption('readOnly', false);
          }
        } catch (_) {}
        if (active?.closest?.('.CodeMirror') || cm.hasFocus?.()) {
          return {
            mode: 'CODEMIRROR_5',
            description: 'CodeMirror 5 Editor',
            element: active,
            container: cmContainer,
            instance: cm
          };
        }
      }
      const cmContent = cmContainer.querySelector('.cm-content') || cmContainer;
      if (active?.closest?.('.cm-editor') || active?.closest?.('.cm-content')) {
        return {
          mode: 'CODEMIRROR_6',
          description: 'CodeMirror 6 Editor',
          element: active,
          container: cmContainer,
          content: cmContent
        };
      }
    }

    const termContainer = active?.closest?.('.xterm, .terminal, [class*="terminal"]');
    if (termContainer) {
      return {
        mode: 'TERMINAL',
        description: 'Integrated Terminal',
        element: active,
        container: termContainer

    // plain textarea or input
    if (active && (active.tagName === 'TEXTAREA' || active.tagName === 'INPUT' || active.isContentEditable)) {
      if (!active.classList?.contains('ace_text-input')) {
        return {
          mode: 'PLAIN_TEXTBOX',
          description: 'Normal Text Box / Textarea',
          element: active
        };
      }

      mode: 'PAGE_CONTAINER',

  // parse katex math formulas into clean readable text
  const extractCleanSelectionText = (win = window) => {
    const sel = win.getSelection();
    if (!sel || sel.rangeCount === 0 || sel.isCollapsed) return '';

    try {
      const range = sel.getRangeAt(0);
      const clone = range.cloneContents();

      // Remove watermark text if selected
      clone.querySelectorAll('[class*="watermark"], [class*="_watermark_"]').forEach(el => el.remove());

      // Format KaTeX formulas
      clone.querySelectorAll('.katex, .katex-display, math, mjx-container, [class*="katex"]').forEach(k => {
        if (!k.parentNode) return;
        const ann = k.querySelector('annotation[encoding="application/x-tex"]') || k.querySelector('annotation');
        let mathText = ann?.textContent ? ann.textContent.trim() : (k.querySelector('.katex-mathml')?.textContent || k.textContent || '');

        const latexMap = {
          '\\\\to': '→', '\\\\rightarrow': '→', '\\\\leftarrow': '←', '\\\\gets': '←',
          '\\\\le': '≤', '\\\\leq': '≤', '\\\\ge': '≥', '\\\\geq': '≥',
          '\\\\ne': '≠', '\\\\neq': '≠', '\\\\times': '×', '\\\\cdot': '·',
          '\\\\pm': '±', '\\\\dots': '...', '\\\\ldots': '...', '\\\\cdots': '...',
          '\\\\min': 'min', '\\\\max': 'max', '\\\\gcd': 'gcd', '\\\\log': 'log',
          '\\\\bmod': 'mod', '\\\\mod': 'mod', '\\\\left': '', '\\\\right': '',
          '\\\\{': '{', '\\\\}': '}', '\\\\_': '_'
        };
        for (const cmd in latexMap) {
          mathText = mathText.split(cmd).join(latexMap[cmd]);
        }
        mathText = mathText.replace(/\\(?:text|mathit|mathbf|mathrm|textrm)\{([^}]+)\}/g, '$1');
        mathText = mathText.replace(/\\([a-zA-Z]+)/g, '$1').trim();

        const replacement = win.document.createTextNode(mathText ? (' ' + mathText + ' ') : ' ');
        if (k.parentNode) k.parentNode.replaceChild(replacement, k);
      });

      clone.querySelectorAll('.katex-html, [aria-hidden="true"], .katex-mathml').forEach(el => el.remove());

      const container = win.document.createElement('div');
      container.style.cssText = 'position:fixed;top:0;left:-9999px;opacity:0;pointer-events:none;z-index:-1;';
      (win.document.body || win.document.documentElement).appendChild(container);
      container.appendChild(clone);
      let text = container.innerText || container.textContent || '';
      container.remove();

      text = text.replace(/\\r\\n/g, '\\n').replace(/\\r/g, '\\n');
      text = text.replace(/\\u00a0/g, ' ').replace(/\\t/g, ' ');
      text = text.split('\\n')
        .map(line => line.replace(/[ ]+/g, ' ').trimEnd())
        .filter((line, idx, arr) => !(line === '' && arr[idx - 1] === ''))
        .join('\\n')
        .trim();

      if (text.length > 0) return text;
    } catch (_) {}

    return sel.toString() || '';
  };

  // copy handlers for ctrl+c and native copy events
  const handleAntidoteCopy = async (win = window) => {
    // Priority 1: User selected text in problem area / statement outside code editors
    const sel = win.getSelection();
    const anchorEl = sel?.anchorNode?.nodeType === 1 ? sel.anchorNode : sel?.anchorNode?.parentElement;
    const isInsideEditor = !!anchorEl?.closest?.('.monaco-editor, .ace_editor, .CodeMirror, .cm-editor');
    const cleanSel = extractCleanSelectionText(win);

    if (!isInsideEditor && cleanSel && cleanSel.length > 0) {
      await safeWriteClipboard(cleanSel);
      notifyCopy();
      return true;
    }

    const state = resolveActiveState(win);
    let text = '';

    try {
      if (state.mode === 'ACE_EDITOR' && state.instance) {
        text = typeof state.instance.getCopyText === 'function' ? state.instance.getCopyText() : (state.instance.getSelectedText() || '');
        if (!text && state.container) {
          const sel = win.getSelection();
          if (sel && !sel.isCollapsed && state.container.contains(sel.anchorNode)) {
            text = sel.toString();
          }
        }
      } else if (state.mode === 'MONACO_VSCODE') {
        const fullSource = await fetchVscodeFile(win);
        const ed = state.instance || (typeof monaco !== 'undefined' && monaco.editor?.getEditors?.()[0]);
        if (ed && typeof ed.getSelection === 'function' && typeof ed.getModel === 'function') {
          const sel = ed.getSelection();
          if (sel && !sel.isEmpty()) text = ed.getModel().getValueInRange(sel);
        }
        const el = state.element;
        if (!text && el && typeof el.selectionStart === 'number' && typeof el.selectionEnd === 'number' && el.selectionStart !== el.selectionEnd) {
          const val = el.value.substring(el.selectionStart, el.selectionEnd);
          if (val) text = expandEllipsisText(val, fullSource);
        }
      } else if (state.mode === 'CODEMIRROR_5' && state.instance) {
        text = state.instance.getSelection() || '';
      } else if (state.mode === 'CODEMIRROR_6') {
        const sel = win.getSelection();
        if (sel && !sel.isCollapsed && state.container?.contains(sel.anchorNode)) {
          text = sel.toString();
        }
      } else if (state.mode === 'TERMINAL') {
        if (typeof window.workbench?.commands?.executeCommand === 'function') {
          window.workbench.commands.executeCommand('workbench.action.terminal.copySelection').then(() => notifyCopy()).catch(() => {});
          return true;
        }
      } else if (state.mode === 'PLAIN_TEXTBOX' && state.element) {
        const el = state.element;
        if (typeof el.selectionStart === 'number' && typeof el.selectionEnd === 'number' && el.selectionStart !== el.selectionEnd) {
          text = el.value.substring(el.selectionStart, el.selectionEnd);
        }
      } else {
        text = extractCleanSelectionText(win);
      }

      if (text && text.length > 0) {
        await safeWriteClipboard(text);
        notifyCopy();
        return true;
      }
      return false;
    } catch (err) {
      console.warn('antidote exception: copy failed', err?.message || err);
      return false;
    }
  };

  const collectBoilerplateFiles = async (win = window) => {

          if (code) {

          }

      } catch (_) {}

      try {

          const scheme = m.uri?.scheme || '';

          const fName = p.split('/').pop() || p;
          if (!fName || fName.startsWith('.') || fName.endsWith('.log')) continue;
          if (!existingPaths.has(p) && !existingPaths.has(fName)) {
            const code = m.getValue();
            if (code && code.trim().length > 0) {
              files.push({ name: fName, path: p, code: code });
              existingPaths.add(p);
            }
          }
        }
      } catch (_) {}
    }

    // Iframe-nested Monaco fallback
    if (files.length === 0) {
      try {
        const iframes = doc.querySelectorAll('iframe');
        for (let i = 0; i < iframes.length; i++) {
          const iwin = iframes[i].contentWindow;
          if (iwin && typeof iwin.monaco !== 'undefined' && iwin.monaco.editor?.getModels) {
            const models = iwin.monaco.editor.getModels();
            for (const m of models) {
              const p = m.uri?.path || '';
              const scheme = m.uri?.scheme || '';
              if (scheme === 'output' || scheme === 'vscode-userdata' || scheme === 'inmemory') continue;
              const fName = p.split('/').pop() || p;
              if (!fName || fName.startsWith('.')) continue;
              const code = m.getValue();
              if (code && code.trim().length > 0) {
                files.push({ name: fName, path: p, code: code });
              }
            }
            if (files.length > 0) break;
          }
        }
      } catch (_) {}
    }

    // Ace Editor (single / multi tab)
    if (files.length === 0) {
      const aceContainer = doc.querySelector('.ace_editor, #submit-ide-v2');
      if (aceContainer) {
        let ed = aceContainer.env?.editor;
        if (!ed && typeof ace !== 'undefined') {
          try { ed = ace.edit(aceContainer); } catch (_) {}
        }
        if (ed) {
          const code = ed.getValue();
          if (code) {
            const langEl = doc.querySelector('[class*="LanguageSelect"], .language-picker, select');
            const lang = (langEl?.innerText || langEl?.value || '').toLowerCase();
            let ext = 'txt';
            if (lang.includes('java')) ext = 'java';
            else if (lang.includes('cpp') || lang.includes('c++')) ext = 'cpp';
            else if (lang.includes('python') || lang.includes('py')) ext = 'py';
            else if (lang.includes('c')) ext = 'c';
            else if (lang.includes('javascript') || lang.includes('js')) ext = 'js';
            files.push({ name: 'Solution.' + ext, code: code });
          }
        }
      }
    }

    // CodeMirror fallback
    if (files.length === 0) {
      const cmContainer = doc.querySelector('.CodeMirror, .cm-editor');
      if (cmContainer?.CodeMirror) {
        const code = cmContainer.CodeMirror.getValue();
        if (code) files.push({ name: 'Solution.txt', code: code });
      }
    }

    return files;
  };

  const generatePromptBundle = async (win = window) => {
    const questionText = extractQuestionStatement(win.document);
    const files = await collectBoilerplateFiles(win);

    let output = 'question area:\n' + (questionText || '(No question statement detected)') + '\n\n';

    if (files.length > 0) {
      const fileNames = files.map(f => f.name).join(', ');
      output += 'boilerplate code files: ' + fileNames + '\n\n';

      const fileBlocks = files.map(f => {
        let commentChar = '// ';
        let commentEnd = '';
        if (f.name.endsWith('.py') || f.name.endsWith('.sh') || f.name.endsWith('.r')) {
          commentChar = '# ';
        } else if (f.name.endsWith('.html') || f.name.endsWith('.xml')) {
          commentChar = '<!-- ';
          commentEnd = ' -->';
        } else if (f.name.endsWith('.sql')) {
          commentChar = '-- ';
        } else if (f.name.endsWith('.css')) {
          commentChar = '/* ';
          commentEnd = ' */';
        }
        return commentChar + f.name + commentEnd + '\n' + f.code.trim();
      });

      output += fileBlocks.join('\n\n\n\n');
    } else {
      output += 'boilerplate code files: none detected';
    }

    return output;
  };

  const showToast = (symbol, text) => {
    let el = document.getElementById('antidote-toast');
    if (!el) {
      el = document.createElement('div');
      el.id = 'antidote-toast';
      (document.body || document.documentElement).appendChild(el);
    }
    const colors = { '✔': '#41A300', '✘': '#e1002d', '!': '#f59e0b', '?': '#C4C4C4' };
    const color = colors[symbol] || '#ffffff';
    el.style.cssText = 'position:fixed;bottom:20px;left:20px;z-index:2147483647;display:flex;align-items:center;gap:8px;padding:6px 11px;background:#18181b;color:#f4f4f5;font-family:ui-monospace,SFMono-Regular,Consolas,monospace;font-size:12px;border:1px solid rgba(255,255,255,0.12);border-radius:6px;box-shadow:0 4px 14px rgba(0,0,0,0.5);pointer-events:none;transition:opacity 0.2s;opacity:1;';
    el.innerHTML = '<span style="color:' + color + ';font-weight:bold">' + symbol + '</span><span>' + text + '</span>';
    clearTimeout(window.__ANTIDOTE_TOAST_TIMER__);
    window.__ANTIDOTE_TOAST_TIMER__ = setTimeout(() => {
      if (el) el.style.opacity = '0';
      setTimeout(() => el?.remove(), 250);
    }, 3000);
  };

  let lastBundleTrigger = 0;
  const triggerPromptBundle = async (win = window) => {
    const now = Date.now();
    if (now - lastBundleTrigger < 2000) return;
    lastBundleTrigger = now;

    const doc = win.document;
    const path = (win.location.pathname || '').toLowerCase();
    const host = (win.location.hostname || '').toLowerCase();

    const isCodeChef = host.includes('codechef.com') || host.includes('localhost') || host.includes('127.0.0.1');
    const isTestRoute = isCodeChef && (
      /(skill-test|problems|submit|ide|contests|practice|learn|assessment|exam|test|quiz|viva|interview|batch|course|series)/i.test(path) ||
      /(skill-test|problems|submit|ide|contests|practice|learn|assessment|exam|test|quiz|viva)/i.test(doc.title || '') ||
      !!doc.querySelector('[class*="Problem"], [class*="Question"], [class*="Viva"], [class*="Mcq"], [class*="Assessment"], [class*="quiz"], #problem-statement, .problem-statement')
    );
    if (!isTestRoute) {
      showToast('✘', 'not on test page');
      return;
    }

    const isLoading = !!doc.querySelector('.loading, .loader-icon-dark, [class*="loadingIcon"], [class*="skeleton"]');
    const questionText = extractQuestionStatement(doc);
    const hasEditor = isCodingPage(doc);
    if (isLoading && !questionText && !hasEditor) {
      showToast('!', 'wait for test to load');
      return;
    }

    try {
      const files = await collectBoilerplateFiles(win);

      if (files && files.length > 0) {
        const bundle = await generatePromptBundle(win);
        await safeWriteClipboard(bundle);
        showToast('✔', 'content copied');
        return;
      }

      if (questionText) {
        const output = 'question area:\n' + questionText;
        await safeWriteClipboard(output);
        showToast('✔', 'question copied');
        return;
      }

      showToast('?', 'no question or code detected');
    } catch (err) {
      console.warn('antidote exception: prompt bundling failed', err?.message || err);
      showToast('✘', 'copy failed');
    }
  };

  function handleAntidoteShortcut(e, win = window) {
    if ((e.ctrlKey || e.metaKey) && e.altKey && (e.key === 'c' || e.key === 'C' || e.code === 'KeyC')) {
      e.preventDefault();
      e.stopPropagation();
      console.log('magic happened');
      triggerPromptBundle(win);
      return true;
    }
    return false;
  }

  // Minimal console scorecard verification
  const isMonaco = !!document.querySelector('.monaco-editor, .monaco-workbench') || typeof monaco !== 'undefined' || typeof window.workbench !== 'undefined' || typeof window.require === 'function';
  const isAce = !!document.querySelector('.ace_editor, #submit-ide-v2, #markdown-editor');
  const isCm5 = !!document.querySelector('.CodeMirror');
  const isCm6 = !isCm5 && !!document.querySelector('.cm-editor');
  const isVerifiedTa = !!document.querySelector('[class*="ShortAnswer"] textarea, [class*="short-answer"] textarea, .short-answer-input');
  const hasUnknownEditor = !isMonaco && !isAce && !isCm5 && !isCm6 && !isVerifiedTa && !!document.querySelector('[class*="editor"], [id*="editor"], textarea, [contenteditable="true"]');

  let detectedEd = 'No';
  if (isMonaco) {
    detectedEd = 'VS Code / Monaco';
  } else if (isAce) {
    detectedEd = 'Ace';
  } else if (isCm5) {
    detectedEd = 'CodeMirror 5';
  } else if (isCm6) {
    detectedEd = 'CodeMirror 6';
  } else if (isVerifiedTa) {
    detectedEd = 'Short Answer Textarea';
  }

  // In-memory kill switch
  window.destroyAntidote = () => {
    if (window.__ANTIDOTE_TOAST_TIMER__) {
      clearTimeout(window.__ANTIDOTE_TOAST_TIMER__);
      delete window.__ANTIDOTE_TOAST_TIMER__;
    }
    if (window.__ANTIDOTE_ROOT_TIMER__) {
      clearInterval(window.__ANTIDOTE_ROOT_TIMER__);
      delete window.__ANTIDOTE_ROOT_TIMER__;
    }
    if (window.__ANTIDOTE_OBSERVER__) {
      try { window.__ANTIDOTE_OBSERVER__.disconnect(); } catch (_) {}
      delete window.__ANTIDOTE_OBSERVER__;
    }
    if (window.__ANTIDOTE_ABORT__) {
      try { window.__ANTIDOTE_ABORT__.abort(); } catch (_) {}
      delete window.__ANTIDOTE_ABORT__;
    }
    document.getElementById('antidote-toast')?.remove();
    document.getElementById('codechef-prompt-bundle-btn')?.remove();
    document.getElementById('codechef-antidote-styles')?.remove();
    document.getElementById('unlock-styles')?.remove();
    document.getElementById('antidote-sandbox')?.remove();
    delete window.__CODECHEF_ANTIDOTE_INITIALIZED__;
    delete window.destroyAntidote;
    console.log('%c[Antidote] Self-destructed. Page restored to default lockdown.', 'color: #ef4444; font-weight: bold;');
  };

  if (hasUnknownEditor) {
    console.warn("Can't help as an unknown editor or text area is detected, wait for updates or try another way.");
  } else {
    console.log(detectedEd + ' code editor found in the dom');
    console.log('Supressed Clipboard blocker, Copy/Paste/Section Acitve Now');
    console.log('it worked buddy, go do your work');
  }
})();
